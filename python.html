Question 1a: 3D Surface Plot

Copy the code below:
import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D

# Data sets as given in the question
x = np.outer(np.linspace(-3, 3, 32), np.ones(32))
y = x.copy().T # transpose
z = (np.sin(x ** 2) + np.cos(y ** 2))

# Create the plot with specified figure size
fig = plt.figure(figsize=(14, 9))
ax = plt.axes(projection='3d')

# Plot the surface
ax.plot_surface(x, y, z, cmap='viridis')

# Labels and Title
ax.set_title('3D Plot Assignment')
ax.set_xlabel('X axis')
ax.set_ylabel('Y axis')
ax.set_zlabel('Z axis')

# Saving the figure (as requested)
plt.savefig("YourMatricNumber_3Dplot.png")
plt.show()

import numpy as np
import matplotlib.pyplot as plt

# Define the x range
x = np.linspace(-10, 10, 400)

# --- Part (i): Plot y = sin(3*x)/x ---
y1 = np.sin(3 * x) / x
plt.figure()
plt.plot(x, y1)
plt.title("Plot 1: y = sin(3x)/x")
plt.savefig("YourMatricNumber_MFP-plot1.png")
plt.show()

# --- Part (ii) & (iii): Varying n (e.g., n=5) with dashed lines ---
n = 5 # You can change n to vary the function
y2 = np.sin(n * x) / x

plt.figure()
plt.plot(x, y2, linestyle='dashed') # specified "dashed lines"
plt.title(f"Plot 2: Varying n (n={n})")

Question 2: Golf Ball Drop Physics

Copy the code below:
# Save file as: YourMatricNumber_golfdropped.py

def calculate_height():
# Constants
g = 9.8 # acceleration due to gravity (m/s^2)

# User Inputs
try:
h_tower = float(input("Enter the height of the tower (in meters): "))
t = float(input("Enter the time interval t (in seconds): "))

# Physics Calculation: s = ut + 0.5gt^2
# Since u (initial velocity) is 0, distance fallen = 0.5 * g * t^2
distance_fallen = 0.5 * g * (t ** 2)

# Calculate height above ground
height_above_ground = h_tower - distance_fallen

# Output
print(f"--- Results ---")
print(f"Height of tower: {h_tower} m")
print(f"Time elapsed: {t} s")
print(f"Height above ground: {height_above_ground:.2f} m")

if height_above_ground < 0:
print("Note: The ball has already hit the ground (Height is 0).")

except ValueError:
print("Please enter valid numbers.")

# Run the function
if __name__ == "__main__":
calculate_height()
